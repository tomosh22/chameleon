# WP-33: /Ob1 /Ob2 of the configuration flags are removed at the end of the target's directory, whatever the call depth.
# The deferred check is queued after the one chameleon_enable queues, so it sees the final flags.
function(_cml_case_check_flags)
    foreach(_cfg RELEASE RELWITHDEBINFO MINSIZEREL)
        if(CMAKE_CXX_FLAGS_${_cfg} MATCHES "[/-]Ob[1-3]")
            message(FATAL_ERROR "case ${CASE}: CMAKE_CXX_FLAGS_${_cfg} still has an /Ob level: ${CMAKE_CXX_FLAGS_${_cfg}}")
        endif()
    endforeach()
    # a plain sibling keeps the level the configuration gave it (as a target option); the enabled target gets /Ob0 only
    get_target_property(_plain plain COMPILE_OPTIONS)
    get_target_property(_gm gm COMPILE_OPTIONS)
    if(NOT _plain MATCHES "RELWITHDEBINFO>:/Ob1>")
        message(FATAL_ERROR "case ${CASE}: the plain sibling lost its /Ob1 in RelWithDebInfo: ${_plain}")
    endif()
    if(_gm MATCHES "RELWITHDEBINFO>:/Ob1>")
        message(FATAL_ERROR "case ${CASE}: the enabled target kept /Ob1: ${_gm}")
    endif()
    message(STATUS "CML-CASE flags without /Ob1 /Ob2, sibling keeps its level: ok")
endfunction()
if(CASE STREQUAL "helper_function")
    function(add_game_module name)
        add_executable(${name} "${CMAKE_BINARY_DIR}/main.cpp")
        chameleon_enable(${name} NO_AGENT)
    endfunction()
    add_executable(plain "${CMAKE_BINARY_DIR}/main.cpp")
    add_game_module(gm)
    cmake_language(DEFER CALL _cml_case_check_flags)
    return()
endif()
if(CASE STREQUAL "child_for_parent")
    # an out-of-tree subdirectory enables a target of its parent: the parent is still open, so no CML1712
    add_executable(parentapp "${CMAKE_BINARY_DIR}/main.cpp")
    add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../project_ext" "${CMAKE_BINARY_DIR}/ext")
    return()
endif()
if(CASE STREQUAL "parent_for_subdir")
    add_subdirectory(sub)
    chameleon_enable(subapp NO_AGENT)  # sub/ is finished: CML1712
    return()
endif()

    set_property(TARGET app PROPERTY INTERPROCEDURAL_OPTIMIZATION ON)

    add_compile_options(/guard:cf)  # directory property only: app was created before
endif()

    chameleon_enable(app)
